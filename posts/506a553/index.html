【信息】题解-洛谷P2014（选课）

发表于 2026-08-13 分类于 information-technology 阅读次数： 本文字数： 183 阅读时长 ≈ 1 分钟

题目大意

课程之间存在先修关系，构成森林。要选 MMM 门课，选一门课必须选它的直接先修课。求最大总学分。

思路

树形DP即可

AC CODE

#include<bits/stdc++.h>
using namespace std;

const int N=305;
vector<int> g[N];
int dp[N][N];
int sz[N];
int val[N];
int n,m;

void dfs(int u,int fa){
    sz[u]=1;
    dp[u][1]=val[u]; //必须选自己
    for(int v:g[u]){
        if(v==fa) continue;
        dfs(v,u);
        //树上背包倒序
        for(int j=sz[u];j>=1;j--){
            for(int t=1;t<=sz[v];t++){
                dp[u][j+t]=max(dp[u][j+t],dp[u][j]+dp[v][t]);
            }
        }
        sz[u]+=sz[v];
    }
}

int main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++){
        int k;
        cin>>k>>val[i];
        g[k].push_back(i);
    }
    dfs(0,-1);
    cout<<dp[0][m];
    return 0;
}
